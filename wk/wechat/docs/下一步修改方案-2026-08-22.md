### A-0 设计
- **不引入 LLM 依赖**。RFM 由现有 `RedemptionRecord`+`Card`+`Customer` 在 SQL 层聚合。
- R（Recency）：顾客最近一次核销距今天数。

### A-2 服务 `services/rfm.py`（新文件）

```python
"""P3-4 RFM 客群分层（本地计算，无需 LLM）。

用 RedemptionRecord + Card + Customer 聚合每位顾客的 R/F/M，
按分位数打 1-5 分，组合成标签。仅读取，不改资金链路。
"""
from datetime import datetime, date
from sqlalchemy import select, func, desc
from ..db import get_session
from ..models import RedemptionRecord, Card, Customer, Merchant

# 标签规则（rfmCell → 中文标签）。可按业务调整。
_LABELS = {
    # R 高(4-5) 为主
    "high_value": "高价值活跃",
    "loyal": "忠实常客",
    "potential": "潜力新客",
    "new": "新客",
    # R 低(1-2) 为主
    "at_risk": "有流失风险",
    "churning": "即将流失",
    "churned": "已流失/沉睡",
    "lost_high": "流失的高价值客（重点挽回）",
}


def _percentile_thresholds(values, qs=(0.2, 0.4, 0.6, 0.8)):
    """返回 4 个分位阈值，用于把值映射到 1-5 档。values 需已排序。"""
    if not values:
        return [0, 0, 0, 0]
    n = len(values)
    out = []
    for q in qs:
        idx = max(0, min(n - 1, int(round(q * n)) - 1))
        out.append(values[idx])
    return out


def _score(value, thresholds, higher_better=True):
    """按阈值把 value 映射 1-5。higher_better=False 时（如 R 距今天数越大越差）反向。"""
    t = thresholds
    if higher_better:
        if value <= t[0]: return 1
        if value <= t[1]: return 2
        if value <= t[2]: return 3
        if value <= t[3]: return 4
        return 5
    else:
        if value <= t[0]: return 5
        if value <= t[1]: return 4
        if value <= t[2]: return 3
        if value <= t[3]: return 2
        return 1


def _label(r, f, m):
    """R/F/M 各 1-5 → 标签。"""
    if r >= 4 and f >= 4 and m >= 4:
        return "高价值活跃"
    if r >= 4 and f >= 3:
        return "忠实常客"
    if r >= 4 and f <= 2:
        return "新客/潜力"
    if r <= 2 and f >= 4 and m >= 4:
        return "流失的高价值客（重点挽回）"
    if r <= 2 and f >= 3:
        return "有流失风险"
    if r <= 2 and f <= 2:
        return "已流失/沉睡"
    if r == 3:
        return "一般活跃"
    return "普通"


def compute_segments(session, *, merchant_id, top=100):
    """计算该商户所有顾客的 RFM，返回按 rfmCell 降序的列表。

    纯读取，不写库。如需快照，调用方再批量 upsert CustomerSegment。
    """
    now = datetime.utcnow()
    # 1) 聚合每位顾客的 F（笔数）、M（储值金额和）、最近核销时间
    rows = session.execute(
        select(
            Card.ownerCustomerId,
            func.count(RedemptionRecord.id).label("freq"),
            func.coalesce(func.sum(RedemptionRecord.amountCents), 0).label("mon"),
            func.max(RedemptionRecord.createdAt).label("last"),
        ).join(Card, Card.id == RedemptionRecord.cardId)
        .where(Card.merchantId == merchant_id,
               RedemptionRecord.revoked == False,  # noqa: E712
               Card.ownerCustomerId.isnot(None))
        .group_by(Card.ownerCustomerId)
    ).all()

    if not rows:
        return {"items": [], "total": 0, "merchantId": merchant_id}

    # 2) 计算 R（距今天数）、收集 F/M 列表用于分位
    recencies, freqs, mons = [], [], []
    items = []
    for cid, freq, mon, last in rows:
        r_days = (now - last).days if last else 9999
        recencies.append(r_days)
        freqs.append(freq)
        mons.append(mon)
        items.append({"customerId": cid, "recencyDays": r_days,
                      "frequency": freq, "monetaryCents": int(mon)})

    recencies.sort()
    freqs.sort()
    mons.sort()
    rt = _percentile_thresholds(recencies)   # R: 越小越好 → higher_better=False
    ft = _percentile_thresholds(freqs)
    mt = _percentile_thresholds(mons)

    # 3) 打分 + 标签
    for it in items:
        r = _score(it["recencyDays"], rt, higher_better=False)
        f = _score(it["frequency"], ft, higher_better=True)
        m = _score(it["monetaryCents"], mt, higher_better=True)
        it["rScore"], it["fScore"], it["mScore"] = r, f, m
        it["rfmCell"] = f"{r}{f}{m}"
        it["label"] = _label(r, f, m)

    # 4) 补顾客名，按 R 降序（最活跃在前）
    cust_ids = [it["customerId"] for it in items]
    custs = session.execute(
        select(Customer).where(Customer.id.in_(cust_ids))
    ).scalars().all()
    cmap = {c.id: c for c in custs}
    for it in items:
        c = cmap.get(it["customerId"])
        it["nickname"] = c.nickname if c else None
        it["phone"] = c.phone if c else None

    items.sort(key=lambda x: (x["rScore"], x["fScore"], x["mScore"]), reverse=True)
    items = items[:top]
    return {"items": items, "total": len(items), "merchantId": merchant_id}


def segment_summary(session, *, merchant_id):
    """客群分层汇总：各标签的人数占比，供仪表盘。"""
    seg = compute_segments(session, merchant_id=merchant_id, top=10000)
    from collections import Counter
    c = Counter(it["label"] for it in seg["items"])
    total = sum(c.values()) or 1
    return {
        "merchantId": merchant_id,
        "totalCustomers": len(seg["items"]),
        "segments": [{"label": k, "count": v, "pct": round(v / total * 100, 1)}
                     for k, v in c.most_common()],
    }
```

### C-2 关键代码骨架 `services/order.py`（新文件）

```python
"""订单/支付服务：购买次卡→支付→自动发卡；续费→支付→延期。

复用 card_svc.issue_card / extend_card，资金链路仍由 RedemptionRecord/CardLedger 权威。
"""
from sqlalchemy import select
from ..errors import BizError, NotFound, Conflict
from ..models import Order, PaymentRecord, Card, CardTemplate, Customer
from . import atomic, lock, audit, save_idempotency, get_idempotency
from . import card as card_svc


def create_order(session, *, merchant_id, customer_id, template_id=None, card_id=None,
                 idempotency_key=None):
    """顾客下单：type=buy_card(需 templateId) 或 renew(需 cardId)。"""
    if idempotency_key:
        cached = get_idempotency(session, idempotency_key)
        if cached is not None:
            return cached
    if template_id:
        tpl = session.get(CardTemplate, template_id)
        if not tpl or tpl.merchantId != merchant_id:
            raise NotFound("卡模板不存在")
        otype, amount = "buy_card", tpl.priceCents
    elif card_id:
        card = session.get(Card, card_id)
        if not card or card.ownerCustomerId != customer_id:
            raise NotFound("卡不存在或非本人")
        tpl = session.get(CardTemplate, card.templateId)
        otype, amount = "renew", tpl.priceCents if tpl else 0
    else:
        raise BizError("BAD_REQUEST", "需指定 templateId 或 cardId")
    o = Order(merchantId=merchant_id, customerId=customer_id, type=otype,
              refCardId=card_id, refTemplateId=template_id,  # 需 Order 加 refTemplateId 列（见下）
              amountCents=amount, status="pending")
    session.add(o); session.flush()
    audit(session, merchant_id=merchant_id, actor_id=customer_id, action="create_order",
          entity_type="order", entity_id=o.id, detail={"type": otype, "amount": amount})
    res = {"orderId": o.id, "type": otype, "amountCents": amount, "status": "pending"}
    if idempotency_key:
        save_idempotency(session, idempotency_key, res)
    return res


def pay_order(session, *, order_id, channel="wechat", transaction_id=None,
              idempotency_key=None):
    """支付成功 → 标记 Order.paid + PaymentRecord.success → 触发发卡/续费。"""
    if idempotency_key:
        cached = get_idempotency(session, idempotency_key)
        if cached is not None:
            return cached
    o = session.execute(lock(select(Order).where(Order.id == order_id))).scalar_one_or_none()
    if not o:
        raise NotFound("订单不存在")
    if o.status == "paid":
        return {"orderId": o.id, "status": "paid", "alreadyPaid": True}
    if o.status not in ("pending",):
        raise Conflict(f"订单状态不可支付：{o.status}")
    pr = PaymentRecord(orderId=o.id, amountCents=o.amountCents, channel=channel,
                       transactionId=transaction_id, success=True)
    session.add(pr); session.flush()
    o.status = "paid"
    # 触发发卡/续费（复用既有幂等链路）
    if o.type == "buy_card" and o.refTemplateId:
        result = card_svc.issue_card(session, merchant_id=o.merchantId,
                                     template_id=o.refTemplateId,
                                     customer_id=o.customerId,
                                     effective_mode="immediate")  # 需 issue_card 支持 customer_id 直发
        o.refCardId = result.get("cardId") if isinstance(result, dict) else result.id
    elif o.type == "renew" and o.refCardId:
        card_svc.extend_card(session, card_id=o.refCardId, merchant_id=o.merchantId,
                             operator_id=o.customerId, add_days=_valid_days_of(session, o.refCardId))
    audit(session, merchant_id=o.merchantId, actor_id=o.customerId, action="pay_order",
          entity_type="order", entity_id=o.id, detail={"channel": channel, "txn": transaction_id})
    res = {"orderId": o.id, "status": "paid", "cardId": o.refCardId}
    if idempotency_key:
```

## 4. 任务 D：退费规则结构化 + 应退计算

### D-0 现状
`CardTemplate.refundRule` 是 `Text` 自由文本，`CardRefundRecord` 有字段但无计算逻辑。P1 合规要"退费规则明示 + 可计算"。

### D-1 设计
- 新增 `RefundRule` 模型（结构化）：按已用次数/时间/余额计算应退。
  - `ruleType`: unused_full（未使用全额）/ prorata_by_used（按已用比例）/ fixed_per_used（按已用单次原价扣）/ no_refund
  - `deductPerUseCents`: 按次扣减额（固定单次原价）
  - `maxRefundCents`: 上限
- 在 `card_customer_detail` 与 `export.build_voucher_data` 输出 `refundEstimateCents`。
- 端点 `GET /api/cards/<id>/refund-estimate`（顾客端/商户 view_refund）。

### D-2 计算函数（`services/refund.py` 新文件）
```python
from sqlalchemy import select, func
from ..models import Card, CardTemplate, RedemptionRecord, CardRefundRecord, Order

def estimate_refund(session, *, card_id):
    """按模板退费规则 + 已核销情况计算应退金额（分）。"""
    card = session.get(Card, card_id)
    if not card: from ..errors import NotFound; raise NotFound("卡不存在")
    tpl = session.get(CardTemplate, card.templateId)
    paid = session.execute(
        select(func.coalesce(func.sum(Order.paidAmount), 0))
        .where(Order.refCardId == card_id, Order.status == "paid")).scalar() or 0
    kind = getattr(card, "cardKind", "times") or "times"
    used = card.usedTimes or 0
    used_cents = session.execute(
        select(func.coalesce(func.sum(RedemptionRecord.amountCents), 0))
        .where(RedemptionRecord.cardId == card_id,
               RedemptionRecord.revoked == False)).scalar() or 0  # noqa: E712

    rule = getattr(tpl, "refundRuleType", None) or "unused_full"
    if rule == "no_refund":
        refund = 0
    elif rule == "prorata_by_used":
        refund = int(paid * max(0, 1 - used / (card.totalTimes or 1))) if kind == "times" \
                 else max(0, (card.faceValueCents or 0) - used_cents)
    elif rule == "fixed_per_used":
        refund = max(0, paid - used * (tpl.deductPerUseCents or 0))
    else:  # unused_full
        refund = (card.faceValueCents or paid) - used_cents if kind == "stored" \
                 else max(0, paid - used * 0)  # 次数卡未使用次数按原价退（简化）
    cap = getattr(tpl, "maxRefundCents", None)
    if cap: refund = min(refund, cap)
    return {"cardId": card_id, "paidCents": int(paid), "usedTimes": used,
            "usedCents": int(used_cents), "refundEstimateCents": max(0, int(refund)),
            "refundRuleType": rule,
            "refundRuleText": tpl.refundRule if tpl else None}
```

> **前置小改**：`CardTemplate` 加 `refundRuleType = Column(String(24))` + `deductPerUseCents = Column(Integer, default=0)` + `maxRefundCents = Column(Integer)`（均可空，migrate 补列）。seed 的模板可补 `refundRuleType="prorata_by_used"`。
